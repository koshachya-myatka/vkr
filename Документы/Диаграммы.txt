@startuml
skinparam actorStyle Hollow
title Просмотр информации о партии металла и формирование отчёта
actor "Пользователь" as User
skinparam defaultFontSize 22
skinparam title {
  fontSize 18
}
skinparam sequence {
  ParticipantFontSize 20
}

participant "Frontend" as Frontend
participant "Backend" as Backend
participant "JWT Filter" as JWT
database "PostgreSQL" as DB
participant "Report Service" as PDF

== Авторизация ==

User -> Frontend : Ввод логина и пароля
activate Frontend

Frontend -> Backend : POST /api/auth/login
activate Backend

Backend -> JWT : Проверка\nпользователя
activate JWT

JWT --> Backend : JWT-токен
deactivate JWT

Backend --> Frontend : Access token
deactivate Backend

Frontend --> User : Успешная авторизация
deactivate Frontend

== Выбор партии ==

User -> Frontend : Просмотр списка партий
activate Frontend

Frontend -> Backend : POST /api/batches\n(filter)
activate Backend

Backend -> DB : Получить\nсписок партий
activate DB

DB --> Backend : Список партий
deactivate DB

Backend --> Frontend : JSON с партиями
deactivate Backend

Frontend --> User : Отображение списка партий
deactivate Frontend

== Детальная информация ==

User -> Frontend : Выбор партии
activate Frontend

Frontend -> Backend : GET /api/batches/{id}
activate Backend

Backend -> DB : Получить данные партии,\nMES, SCADA, LIMS
activate DB

DB --> Backend : Детальная информация
deactivate DB

Backend --> Frontend : JSON\nс данными партии
deactivate Backend

Frontend --> User : Отображение деталей
deactivate Frontend

== Формирование отчёта ==

User -> Frontend : Сформировать PDF-отчёт
activate Frontend

Frontend -> Backend : POST /api/report/{id}
activate Backend

Backend -> PDF : Генерация PDF
activate PDF

PDF --> Backend : PDF-документ
deactivate PDF

Backend --> Frontend : PDF-файл
deactivate Backend

Frontend --> User : Просмотр отчёта
deactivate Frontend

@enduml

' ===================== ENUMS =====================

enum MetalTypeEnum {
  NI ("Никель (Ni)")
  CU ("Медь (Cu)")
  CO ("Кобальт (Co)")
  AG ("Серебро (Ag)")
  AU ("Золото (Au)")
  PD ("Палладий (Pd)")
  PT ("Платина (Pt)")
  RH ("Родий (Rh)")
  IR ("Иридий (Ir)")
  RU ("Рутений (Ru)")
  --
  - name : String
  --
  + toString() : String
  + fromName(name : String) : MetalTypeEnum
}

enum MesProcessStatusEnum {
  ARRIVAL
  PROCESSING
  ANALYSIS
  ACCEPTED
  DEFECTIVE
  --
  + toString() : String
}

enum MesStatusEnum {
  NORMAL
  WARNING
  ALARM
  --
  + toString() : String
}

enum LimsStatusEnum {
  REJECTED
  APPROVED
  --
  + toString() : String
}

enum ScadaStatusEnum {
  NORMAL
  WARNING
  ALARM
  --
  + toString() : String
}

enum NotificationSeverityEnum {
  INFO
  WARNING
  ALARM
}

enum NotificationStatusEnum {
  CREATED ("СОЗДАНО")
  IN_PROGRESS ("НА ПРОВЕРКЕ")
  FALSE_POSITIVE ("ЛОЖНОЕ СРАБАТЫВАНИЕ")
  RESOLVED ("ПРИЧИНА УСТРАНЕНА")
  --
  - name : String
  --
  + toString() : String
}

enum UserRoleEnum {
  LABORATORY
  PRODUCTION
  MANAGEMENT
  ADMIN
}

' ===================== ENTITIES =====================

class DimBatchEntity {
  - batchId : String  <<PK>>
  - metalType : MetalTypeEnum
  - startTime : LocalDateTime
  - processingTime : LocalDateTime
  - analysesTime : LocalDateTime
  - endTime : LocalDateTime
  - processStatus : MesProcessStatusEnum
  - outputYield : Double
}

class MesEntity {
  - recordId : String  <<PK>>
  - equipmentId : String
  - operatorId : String
  - temperature : Double
  - pressure : Double
  - durationSec : Integer
  - energyConsumption : Double
  - status : MesStatusEnum
}

class LimsEntity {
  - recordId : String  <<PK>>
  - sampleId : String
  - analysisMethod : String
  - testDate : LocalDateTime
  - status : LimsStatusEnum
}

class LimsResultEntity {
  - id : Long  <<PK, generated>>
  - parameterName : String
  - value : String
  - unit : String
  - normal : Boolean
}

class ScadaEntity {
  - recordId : String  <<PK>>
  - sensorId : String
  - equipmentId : String
  - time : LocalDateTime  <<PK>>
  - parameter : String
  - value : Double
  - unit : String
  - status : ScadaStatusEnum
}

class NotificationEntity {
  - id : Long  <<PK, generated>>
  - message : String
  - equipmentId : String
  - sensorId : String
  - signalSource : String
  - severity : NotificationSeverityEnum
  - status : NotificationStatusEnum
  - viewed : Boolean
  - comment : String
  - createdAt : LocalDateTime
  - updatedAt : LocalDateTime
  - updatedBy : String
}

class UserEntity {
  - userId : UUID  <<PK>>
  - username : String
  - password : String
  - name : String
  - surname : String
  - patronymic : String
  - email : String
  - role : UserRoleEnum
  - createdAt : LocalDateTime
}

' ===================== RELATIONS =====================

DimBatchEntity "1" -- "1" MesEntity : содержит
DimBatchEntity "1" -- "М  " LimsEntity : содержит
LimsEntity "1" -- "М  " LimsResultEntity : включает
DimBatchEntity ..> MetalTypeEnum : metalType
DimBatchEntity ..> MesProcessStatusEnum : processStatus
MesEntity ..> MesStatusEnum : status
LimsEntity ..> LimsStatusEnum : status
ScadaEntity ..> ScadaStatusEnum : status
NotificationEntity ..> NotificationSeverityEnum : severity
NotificationEntity ..> NotificationStatusEnum : status
UserEntity .> UserRoleEnum : role

@enduml
